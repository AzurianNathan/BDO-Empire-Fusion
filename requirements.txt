<script>
const $ = (id) => document.getElementById(id);
const fmt = (n) => new Intl.NumberFormat("en-US").format(Math.round(n || 0));
let state = { prices: null, farming: 0, itemCount: 0, job: null, poll: null, result: null, cursor: 0 };

function log(line){
  const el = $("log");
  const div = document.createElement("div");
  div.textContent = new Date().toLocaleTimeString() + "  " + line;
  el.appendChild(div); el.scrollTop = el.scrollHeight;
}
function setJob(text, cls){ const j = $("jobline"); j.textContent = text; j.className = "jobline " + (cls||""); }
function priceReady(count, source){
  state.itemCount = count;
  $("pdot").className = "dot ok";
  $("pstate").textContent = fmt(count) + " items \u00b7 " + source;
  $("tItems").textContent = fmt(count);
  $("optimize").disabled = false;
}

// health
fetch("/api/health").then(r => r.ok ? r.json() : Promise.reject()).then(() => {
  $("conn").textContent = "Backend connected"; $("conn").className = "conn ok"; log("Backend connected.");
}).catch(() => { $("conn").textContent = "Backend offline"; $("conn").className = "conn bad"; });

// settings wiring
$("provider").onchange = () => {
  $("customRow").style.display = $("provider").value === "custom" ? "" : "none";
};
$("farmEnable").onchange = () => {
  $("farmRow").style.display = $("farmEnable").checked ? "" : "none";
  if(!$("farmEnable").checked) $("farming").value = 0;
};

let overrides = { custom: {}, keep: {} };

function priceBody(){
  return {
    region: $("region").value,
    lang: $("lang").value,
    tax: Number($("tax").value),
    keepItems: overrides.keep,
    customPrices: overrides.custom,
    provider: $("provider").value,
    customUrl: $("customUrl").value.trim(),
    apiKey: $("apiKey").value,
  };
}

async function loadPrices(){
  $("fetchPrices").disabled = true;
  log("Fetching prices for " + $("region").value + " (tax " + $("tax").value + ")...");
  $("pdot").className = "dot"; $("pstate").textContent = "Loading...";
  try{
    const r = await fetch("/api/effective-prices", {
      method:"POST", headers:{"Content-Type":"application/json"}, body: JSON.stringify(priceBody())
    });
    if(!r.ok){
      let detail = "server " + r.status;
      try{ const j = await r.json(); if(j.detail) detail = j.detail; }catch(_){}
      throw new Error(detail);
    }
    const d = await r.json();
    state.prices = d.effectivePrices;
    priceReady(d.count, ($("provider").value === "custom" ? "custom" : $("provider").options[$("provider").selectedIndex].text) + " \u00b7 " + d.region);
    renderTable(d.rows);
    $("exportPrices").disabled = false;
    log("Prices ready: " + fmt(d.count) + " items.");
    if(d.source) log("Source: " + d.source);
  }catch(e){
    $("pdot").className = "dot bad"; $("pstate").textContent = "Fetch failed";
    log("Price fetch failed: " + e.message);
  }finally{ $("fetchPrices").disabled = false; }
}
$("fetchPrices").onclick = loadPrices;

function renderTable(rows){
  if(!rows || !rows.length){ $("tableWrap").innerHTML = '<p class="note" style="margin:0">No rows.</p>'; return; }
  const head = '<tr><th>Item</th><th style="text-align:right">Market</th><th>Custom</th><th>Keep</th><th style="text-align:right">Effective</th></tr>';
  const body = rows.map(r =>
    '<tr data-id="' + r.id + '">' +
      '<td>' + r.name + (r.vendor ? '<span class="vtag">vendor</span>' : '') + '</td>' +
      '<td class="num">' + (r.market == null ? "\u2014" : fmt(r.market)) + '</td>' +
      '<td><input type="number" class="cust" min="0" value="' + (r.custom === "" ? "" : r.custom) + '" /></td>' +
      '<td><input type="checkbox" class="keep"' + (r.keep ? " checked" : "") + ' /></td>' +
      '<td class="num eff">' + fmt(r.effective) + '</td>' +
    '</tr>').join("");
  $("tableWrap").innerHTML = '<table><thead>' + head + '</thead><tbody>' + body + '</tbody></table>';
  $("tableWrap").querySelectorAll("tr[data-id]").forEach(tr => {
    const id = tr.dataset.id;
    tr.querySelector(".cust").onchange = (e) => {
      const v = e.target.value;
      if(v === "") delete overrides.custom[id]; else overrides.custom[id] = Number(v);
    };
    tr.querySelector(".keep").onchange = (e) => {
      if(e.target.checked) overrides.keep[id] = true; else delete overrides.keep[id];
    };
  });
}

// import / export / clear
$("fileInput").onchange = (ev) => {
  const file = ev.target.files[0]; if(!file) return;
  const reader = new FileReader();
  reader.onload = () => {
    try{
      const j = JSON.parse(reader.result);
      if(j.customPrices || j.keepItems){
        overrides.custom = j.customPrices || {};
        overrides.keep = j.keepItems || {};
        if(j.tax) $("tax").value = String(j.tax);
        log("Imported overrides: " + Object.keys(overrides.custom).length + " custom, " + Object.keys(overrides.keep).length + " keep. Reloading...");
        loadPrices();
      }else if(j.effectivePrices){
        state.prices = j.effectivePrices;
        if(j.farmingWorkerSilverPerDay){ $("farmEnable").checked = true; $("farmEnable").onchange(); $("farming").value = j.farmingWorkerSilverPerDay; }
        priceReady(Object.keys(j.effectivePrices).length, "import file");
        log("Loaded price export: " + fmt(Object.keys(j.effectivePrices).length) + " items.");
      }else{ throw new Error("no effectivePrices / customPrices field"); }
    }catch(e){ log("Could not read import: " + e.message); }
  };
  reader.readAsText(file);
};

$("exportPrices").onclick = () => {
  const payload = {
    effectivePrices: state.prices,
    farmingWorkerSilverPerDay: Number($("farming").value) || 0,
    tax: Number($("tax").value),
    customPrices: overrides.custom,
    keepItems: overrides.keep,
  };
  const blob = new Blob([JSON.stringify(payload, null, 2)], { type:"application/json" });
  const a = document.createElement("a");
  a.href = URL.createObjectURL(blob); a.download = "prices.json"; a.click();
};

$("cmpBtn").onclick = async () => {
  $("cmpBtn").disabled = true;
  log("Fetching the same items from every source (compare against in-game to spot stale data)...");
  try{
    const r = await fetch("/api/compare-sources/" + $("region").value + "?lang=" + $("lang").value);
    const d = await r.json();
    for(const x of d.results){
      const parts = Object.entries(x.prices).map(([n,v]) => n + "=" + (v==null?"-":fmt(v)));
      log((x.ok?"  ":"  x ") + x.label + " -> " + (parts.length?parts.join("  "):x.detail));
    }
    log("Whichever row matches your in-game prices is the live one; select it above.");
  }catch(e){ log("Compare failed: " + e.message); }
  finally{ $("cmpBtn").disabled = false; }
};

$("clearOverrides").onclick = () => {
  overrides = { custom: {}, keep: {} };
  log("Cleared custom prices and keep flags.");
  if(state.prices) loadPrices();
};

// optimize
$("optimize").onclick = async () => {
  if(!state.prices){ log("Load prices first."); return; }
  $("valNum").textContent = "..."; $("download").disabled = true; state.result = null;
  const budget = Number($("budget").value);
  $("tBudget").textContent = fmt(budget);
  let dropped = 0;
  const clean = {};
  for(const k in (state.prices||{})){
    const v = Number(state.prices[k]);
    if(Number.isFinite(v)) clean[k] = v; else dropped++;
  }
  if(dropped) log(dropped + " item(s) had no usable price and will be valued at 0.");
  const body = {
    budget,
    effectivePrices: clean,
    droppedPrices: dropped,
    farmingWorkerSilverPerDay: $("farmEnable").checked ? (Number($("farming").value) || 0) : 0,
    modifiers: {},
    baseEmpire: null,
    solverOverrides: { mip_rel_gap: Number($("gap").value) || 0.0001 },
  };
  setJob("Starting...", "run"); $("optimize").disabled = true; $("stop").disabled = false;
  log("Optimize starting - budget " + budget + " CP.");
  try{
    const r = await fetch("/api/optimize", { method:"POST", headers:{"Content-Type":"application/json"}, body: JSON.stringify(body) });
    const d = await r.json(); state.job = d.job_id; state.cursor = 0;
    setJob("Solving...", "run"); log("Job " + d.job_id.slice(0,8) + " running. Large budgets can take a while.");
    poll();
  }catch(e){ setJob("Failed to start", "err"); log("Failed: " + e.message); $("optimize").disabled=false; $("stop").disabled=true; }
};

function poll(){
  state.poll = setInterval(async () => {
    try{
      const r = await fetch("/api/optimize/" + state.job + "?since=" + state.cursor);
      const d = await r.json();
      if(d.lines && d.lines.length){
        for(const ln of d.lines) log("  " + ln);
        state.cursor = d.cursor;
      }
      if(d.status === "done" || d.status === "stopped"){ clearInterval(state.poll); finish(d); }
      else if(d.status === "error"){ clearInterval(state.poll); setJob("Solver error", "err"); log("Solver error (see server console)."); reset(); }
    }catch(e){ clearInterval(state.poll); setJob("Lost server", "err"); log("Lost contact: " + e.message); reset(); }
  }, 2000);
}
function finish(d){
  reset();
  if(d.result){
    state.result = d.result;
    const workers = (d.result.userWorkers||[]).length;
    const nodes = (d.result.grindTakenList||[]).length;
    $("tWorkers").textContent = fmt(workers);
    $("tNodes").textContent = fmt(nodes);
    $("valNum").textContent = "ready";
    $("download").disabled = false;
    setJob(d.status === "stopped" ? "Stopped (best so far)" : "Done", "done");
    log("Optimized empire ready: " + workers + " workers. Download and import into Workerman.");
  }else{ setJob("No empire returned", "err"); }
}
function reset(){ $("optimize").disabled = false; $("stop").disabled = true; }

$("stop").onclick = async () => {
  if(!state.job) return;
  log("Stopping - keeps best solution so far.");
  await fetch("/api/optimize/" + state.job + "/stop", { method:"POST" });
};

$("download").onclick = () => {
  if(!state.result) return;
  const blob = new Blob([JSON.stringify(state.result, null, 2)], { type:"application/json" });
  const a = document.createElement("a");
  a.href = URL.createObjectURL(blob); a.download = "optimized_empire.json"; a.click();
};
</script>
